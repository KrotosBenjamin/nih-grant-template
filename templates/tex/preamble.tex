% pandoc/LaTeX template for NIH grant pdfs (minimal)
\usepackage{changepage} % provides the adjustwidth environment
\usepackage{setspace}
\setstretch{1.0} % single-spaced body
\usepackage{geometry}

% Font and margins are set in _quarto.yml (TeX Gyre Heros 11pt, 0.5in margins).
% TeX Gyre Heros approximates Arial; verify font requirements for your FOA.
% To switch to pdflatex instead of lualatex, set `pdf-engine: pdflatex` in
% _quarto.yml, drop `mainfont`, and add here:
%   \usepackage[scaled]{helvet}
%   \renewcommand{\familydefault}{\sfdefault}
\usepackage[hidelinks]{hyperref}
\usepackage{microtype}

% Colors: do NOT \definecolor here. The palette lives in filters/color.lua
% (the PALETTE table) and is injected into this preamble automatically, so the
% two cannot drift apart. Authors write [text]{color="accent"} or
% [text]{color="#B00020"} in markdown.

% Default PDF highlighting, shared by every mechanism. Colors are supplied
% by filters/color.lua and expanded only when headings/quotes are typeset.
\usepackage{titlesec}
\titleformat{\section}
  {\bfseries\large\filcenter\color{heading}}{}{0pt}{\MakeUppercase}
\titlespacing*{\section}{0pt}{0pt}{0.6ex}

% Keep the entire band within the text measure; a parbox lets long titles wrap.
\newcommand{\grantbandhead}[1]{%
  \begingroup
  \setlength{\fboxsep}{3pt}%
  \colorbox{headingbg}{%
    \parbox{\dimexpr\linewidth-2\fboxsep\relax}{\raggedright #1}}%
  \endgroup}
\titleformat{\subsection}[block]
  {\bfseries\normalsize\color{heading}}{}{0pt}{\grantbandhead}
\titlespacing*{\subsection}{0pt}{1.1ex}{0.5ex}
\titleformat{\subsubsection}[block]
  {\bfseries\normalsize\color{heading}}{}{0pt}
  {\rule[-0.15ex]{2.5pt}{1.0em}\hspace{0.5em}}
\titlespacing*{\subsubsection}{0pt}{1.0ex}{0.3ex}
\titleformat{\paragraph}[runin]{\bfseries\normalsize}{}{0pt}{}
\titlespacing*{\paragraph}{0pt}{0.5ex}{1em}

% Markdown > blocks become compact shaded statements. These boxes do not
% split across pages: reserve block quotes for short questions or takeaways.
\usepackage{varwidth}
\newsavebox{\grantquotebox}
\renewenvironment{quote}
  {\par\addvspace{4pt}%
    \begingroup
    \setlength{\fboxsep}{3pt}%
    \setlength{\fboxrule}{0.5pt}%
    \begin{lrbox}{\grantquotebox}%
      \begin{varwidth}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}\centering}
  {   \end{varwidth}%
    \end{lrbox}%
    \noindent\makebox[\linewidth][c]{%
      \fcolorbox{quoterule}{quotebg}{\usebox{\grantquotebox}}}%
    \par\endgroup\addvspace{4pt}}

% Kill page numbers
\AtBeginDocument{\pagenumbering{gobble}\pagestyle{empty}}

% kill hyperlinks only in bibliography
\usepackage{etoolbox}
\AtBeginEnvironment{CSLReferences}{%
  \let\oldhref\href
  \let\oldurl\url
  \providecommand{\doi}[1]{#1}% some CSL styles emit \doi{...}
  \renewcommand{\href}[2]{#2}% keep visible text, drop link
  \renewcommand{\url}[1]{#1}% print URL as plain text
}
\AtEndEnvironment{CSLReferences}{%
  \let\href\oldhref
  \let\url\oldurl
}
